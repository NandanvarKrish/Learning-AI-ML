📖 The Project

Student Grade Management System

Build a complete, production-quality grade management system that a real school could use. It reads and saves data, handles bad input gracefully, uses OOP throughout, and generates reports.

📋Features to implement: Add students (with validation), view ranked leaderboard, search by name, save to CSV, load from CSV, export JSON summary, generate batch reports, support ScholarStudent (inherits Student + scholarship logic).
📁 Part 1 — Student Class
Python — student.py
# student.py — Student class import datetime class Student: """Represents a single student with marks and metadata.""" pass_mark = 50 # class variable — shared def __init__(self, roll, name, marks): self.roll = roll self.name = name.strip().title() self.marks = marks # uses @setter below self.added = datetime.date.today() @property def marks(self): return self._marks @marks.setter def marks(self, val): if not isinstance(val, (int, float)): raise TypeError("marks must be a number") if not (0 <= val <= 100): raise ValueError(f"marks {val} not in 0–100") self._marks = round(val, 1) @property def grade(self): for threshold, letter in [ (90,"A+"),(80,"A"),(70,"B"), (60,"C"),(50,"D") ]: if self._marks >= threshold: return letter return "F" @property def is_passing(self): return self._marks >= Student.pass_mark def to_dict(self): return { "roll" : self.roll, "name" : self.name, "marks" : self._marks, "grade" : self.grade, "status": "Pass" if self.is_passing else "Fail" } def __repr__(self): return f"Student({self.roll}, {self.name}, {self.grade})" class ScholarStudent(Student): """Student who also tracks scholarship status.""" scholarship_threshold = 85 def __init__(self, roll, name, marks, scholarship=False): super().__init__(roll, name, marks) self.scholarship = scholarship @property def qualifies_for_scholarship(self): return self._marks >= ScholarStudent.scholarship_threshold def to_dict(self): d = super().to_dict() # extend parent's to_dict d["scholar"] = self.qualifies_for_scholarship return d
📁 Part 2 — GradeBook Class
Python — gradebook.py
# gradebook.py — GradeBook class import csv, json, os, datetime from student import Student class GradeBook: """Manages a collection of students with persistence.""" def __init__(self, filename="students.csv"): self.filename = filename self.students = self._load() def _load(self): if not os.path.exists(self.filename): return [] students = [] with open(self.filename) as f: for row in csv.DictReader(f): try: s = Student(row["roll"],row["name"],float(row["marks"])) students.append(s) except (ValueError, KeyError) as e: print(f"⚠️ Skipped row: {e}") return students def _save(self): with open(self.filename, "w", newline="") as f: fields = ["roll","name","marks","grade","status"] w = csv.DictWriter(f, fieldnames=fields) w.writeheader() w.writerows(s.to_dict() for s in self.students) def add(self, roll, name, marks): try: s = Student(roll, name, marks) self.students.append(s) self._save() print(f"✅ Added {s.name} — Grade: {s.grade}") except (ValueError, TypeError) as e: print(f"❌ Error: {e}") def search(self, name): query = name.lower() results = [s for s in self.students if query in s.name.lower()] return results @property def stats(self): if not self.students: return {} all_marks = [s.marks for s in self.students] return { "total" : len(self.students), "average": round(sum(all_marks)/len(all_marks), 2), "highest": max(all_marks), "lowest" : min(all_marks), "passed" : sum(1 for s in self.students if s.is_passing), "grades" : {g: sum(1 for s in self.students if s.grade==g) for g in ["A+","A","B","C","D","F"]} } def leaderboard(self, top=5): return sorted(self.students, key=lambda s: s.marks, reverse=True)[:top] def report(self, *args, **kwargs): show_stats = kwargs.get("show_stats", True) sort_by = kwargs.get("sort_by", "marks") print(f"\n{'='*44}") print(f" GRADE REPORT — {datetime.date.today()}") print(f"{'='*44}") ordered = sorted(self.students, key=lambda s: getattr(s, sort_by, s.marks), reverse=True) for rank, s in enumerate(ordered, 1): icon = "✅" if s.is_passing else "❌" print(f" {rank:>2}. {s.name:15} {s.marks:>5} {s.grade:>3} {icon}") if show_stats: st = self.stats print(f"\n Average: {st['average']} | Passed: {st['passed']}/{st['total']}") def batch_report(self, batch_size=3): """Generator — yield report chunks for large classes.""" for i in range(0, len(self.students), batch_size): yield self.students[i:i+batch_size] def export_json(self): data = { "exported": str(datetime.date.today()), "stats" : self.stats, "students": [s.to_dict() for s in self.students] } with open("report.json", "w") as f: json.dump(data, f, indent=4) print("📄 Exported to report.json")
📁 Part 3 — main.py
Python — main.py
# main.py — run the program from gradebook import GradeBook from student import ScholarStudent gb = GradeBook() # loads existing data if file exists # Add students (validation in action) gb.add("001", "ali ahmed", 92) gb.add("002", "sara khan", 45) gb.add("003", "usman malik", 78) gb.add("004", "zara butt", 88) gb.add("005", "bad student", 150) # ← triggers ValueError! # Add a scholar student (inheritance) scholar = ScholarStudent("006", "Hira Baig", 94) gb.students.append(scholar) print(f"Scholarship? {scholar.qualifies_for_scholarship}") # Print full report gb.report(show_stats=True, sort_by="marks") # Leaderboard (top 3) print("\n🏆 TOP 3:") for i, s in enumerate(gb.leaderboard(3), 1): print(f" #{i} {s.name}: {s.marks}") # Search print("\n🔍 Search 'ali':") for s in gb.search("ali"): print(f" {s}") # Stats print(f"\n📊 Class stats: {gb.stats}") # Batch generator print("\n📋 Batch report (3 per batch):") for i, batch in enumerate(gb.batch_report(3), 1): print(f" Batch {i}: {[s.name for s in batch]}") # Export JSON gb.export_json()
