Method Chaining — Writing Professional Pandas Code

Readable, functional-style pipelines instead of step-by-step reassignment

Method chaining links multiple Pandas operations in one fluent expression instead of creating intermediate variables at every step. It's the style used in professional codebases and matches how R's dplyr / SQL pipelines read.

PYTHON
# ❌ Step-by-step style — verbose, many intermediate variables
df1 = df.dropna(subset=['Salary'])
df2 = df1[df1['Salary'] > 50000]
df3 = df2.groupby('Dept')['Salary'].mean()
df4 = df3.sort_values(ascending=False)
print(df4)

# ✅ Method chaining — one fluent readable pipeline
result = (
    df
    .dropna(subset=['Salary'])
    .query('Salary > 50000')
    .groupby('Dept')['Salary']
    .mean()
    .sort_values(ascending=False)
)
print(result)

# pipe() — insert a custom function into a chain
def add_bonus_col(df, pct=0.1):
    return df.assign(Bonus = df['Salary'] * pct)

result2 = (
    df
    .dropna(subset=['Salary'])
    .pipe(add_bonus_col, pct=0.15)
    .sort_values('Bonus', ascending=False)
)

# assign() — add multiple derived columns fluently in a chain
result3 = (
    df
    .assign(
        Bonus     = lambda x: x['Salary'] * 0.1,
        TotalComp = lambda x: x['Salary'] + x['Bonus']
    )
    .sort_values('TotalComp', ascending=False)
    .head(10)
)
💡 Why Chain?
  • No intermediate variable clutter (df1, df2, df3...)
  • Reads top-to-bottom like a data pipeline / recipe
  • Easy to comment out or reorder individual steps
  • Matches modern data engineering style (dbt, dplyr, Spark)

Window Functions — Rank, Cumulative, Group Comparisons

SQL-style window functions for ranking and running calculations

Ranking Functions

PYTHON
df = pd.DataFrame({
    'Dept'  :['Eng','Eng','Eng','HR','HR'],
    'Name'  :['Alice','Bob','Carol','Dave','Eve'],
    'Salary':[95000,85000,90000,60000,65000]
})

# Overall rank (1 = highest salary)
df['Rank'] = df['Salary'].rank(ascending=False).astype(int)

# Rank WITHIN each department — the SQL "PARTITION BY" equivalent
df['DeptRank'] = df.groupby('Dept')['Salary'].rank(ascending=False).astype(int)
print(df)
#   Dept   Name  Salary  Rank  DeptRank
# 0  Eng  Alice   95000     1         1
# 1  Eng    Bob   85000     4         3
# 2  Eng  Carol   90000     3         2
# 3   HR   Dave   60000     5         2
# 4   HR    Eve   65000     2         1   ← ranks reset per department!

# Percentile rank
df['PctRank'] = df['Salary'].rank(pct=True)

# Top-N per group — a very common business question
top2_per_dept = (df.sort_values('Salary', ascending=False)
                    .groupby('Dept')
                    .head(2))
print(top2_per_dept)

Cumulative & Group-Relative Functions

PYTHON
# Cumulative sum WITHIN each group
df['CumSalary'] = df.groupby('Dept')['Salary'].cumsum()

# Row number within group (like SQL ROW_NUMBER())
df['RowInDept'] = df.groupby('Dept').cumcount() + 1

# Percentage of group total
df['PctOfDeptTotal'] = df['Salary'] / df.groupby('Dept')['Salary'].transform('sum')

# First/last value in group, broadcast to all rows
df['DeptMaxSalary'] = df.groupby('Dept')['Salary'].transform('max')
df['GapFromTop'] = df['DeptMaxSalary'] - df['Salary']

Performance — Vectorisation over apply() over loops

Write Pandas code that scales to millions of rows

⚡ The Speed Hierarchy
1. Vectorised operations (fastest — C-speed)
2. .apply() with a function (medium — Python loop internally)
3. .iterrows() / for loop (slowest — avoid at all costs)
Always look for a vectorised way first. Reach for apply() only when truly needed.
PYTHON
import time
n = 100_000
df = pd.DataFrame({'Price': np.random.uniform(10,1000,n)})

# ❌ SLOWEST: iterrows() loop
t0 = time.time()
result = []
for idx, row in df.iterrows():
    result.append(row['Price'] * 1.1)
print(f"iterrows: {time.time()-t0:.3f}s")   # ~2-4 seconds!

# 🐢 MEDIUM: apply()
t0 = time.time()
df['Price2'] = df['Price'].apply(lambda x: x * 1.1)
print(f"apply:    {time.time()-t0:.3f}s")     # ~0.02-0.05 seconds

# ✅ FASTEST: vectorised
t0 = time.time()
df['Price3'] = df['Price'] * 1.1
print(f"vector:   {time.time()-t0:.3f}s")     # ~0.0005 seconds — 100-1000x faster!

# More vectorisation examples
# ❌ apply for conditional logic
df['Tier'] = df['Price'].apply(lambda x: 'High' if x > 500 else 'Low')
# ✅ vectorised with np.where
df['Tier'] = np.where(df['Price'] > 500, 'High', 'Low')

# ❌ apply for string ops
df['Upper'] = df['Name'].apply(lambda x: x.upper())
# ✅ vectorised .str accessor
df['Upper'] = df['Name'].str.upper()

Build 4 Related Tables with Realistic Messiness
STARTER CODE
import pandas as pd
import numpy  as np

rng = np.random.default_rng(2025)

# TABLE 1: Stores
stores = pd.DataFrame({
    'StoreID': range(1,9),
    'City'   : ['Mumbai','Delhi','Bangalore','Pune',
                'Chennai','Hyderabad','Kolkata','Jaipur'],
    'Region' : ['West','North','South','West',
                'South','South','East','North']
})

# TABLE 2: Products
products = pd.DataFrame({
    'ProductID': range(101,121),
    'Category' : rng.choice(['Grocery','Electronics','Apparel','Home'], 20),
    'UnitCost' : rng.uniform(50, 15000, 20).round(2)
})
products['UnitPrice'] = (products['UnitCost'] * rng.uniform(1.2,1.8,20)).round(2)

# TABLE 3: Customers (with messy fields)
n_cust = 300
customers = pd.DataFrame({
    'CustID'    : range(1, n_cust+1),
    'Name'      : [f'  Customer_{i} ' for i in range(1,n_cust+1)],
    'SignupDate': pd.to_datetime('2023-01-01') + pd.to_timedelta(rng.integers(0,500,n_cust), unit='D'),
    'Segment'   : rng.choice(['regular','REGULAR','premium','PREMIUM','vip'], n_cust)
})

# TABLE 4: Transactions (messy, 2 years, some invalid refs)
n_txn = 5000
transactions = pd.DataFrame({
    'TxnID'    : range(1, n_txn+1),
    'CustID'   : rng.choice(range(1, n_cust+15), n_txn),  # some invalid CustIDs
    'StoreID'  : rng.choice(range(1,9), n_txn),
    'ProductID': rng.choice(range(101,121), n_txn),
    'Quantity' : rng.choice([1,2,3,4,5,-1], n_txn, p=[0.3,0.25,0.2,0.15,0.05,0.05]),
    'TxnDate'  : pd.to_datetime('2023-01-01') + pd.to_timedelta(rng.integers(0,730,n_txn), unit='D'),
    'PaymentType': rng.choice(['Card','card','CASH','cash','UPI','upi'], n_txn)
})

# Add 100 duplicate transactions
transactions = pd.concat([transactions, transactions.sample(100, random_state=1)], ignore_index=True)

print(f"Stores:{stores.shape} Products:{products.shape} Customers:{customers.shape} Transactions:{transactions.shape}")

Phase 3 — Merge & Feature Engineering

~35 min▼
1
Build the Master Analytical Table
CODE
master = (
    transactions_clean
    .merge(products, on='ProductID', how='left')
    .merge(stores, on='StoreID', how='left')
    .merge(customers_clean, on='CustID', how='left', indicator='cust_status')
    .assign(
        Revenue = lambda x: x['Quantity'] * x['UnitPrice'],
        Profit  = lambda x: x['Quantity'] * (x['UnitPrice'] - x['UnitCost']),
        Segment = lambda x: x['Segment'].fillna('guest')
    )
)
print(f"Master table: {master.shape}")
print(master['cust_status'].value_counts())

Phase 5 — Customer Segmentation & Rankings

~30 min▼
1
Build an RFM-Style Customer Table

Recency, Frequency, Monetary — the classic customer segmentation framework.

CODE
snapshot_date = master['TxnDate'].max() + pd.Timedelta(days=1)

rfm = master.groupby('CustID').agg(
    Recency   = ('TxnDate', lambda x: (snapshot_date - x.max()).days),
    Frequency = ('TxnID', 'count'),
    Monetary  = ('Revenue', 'sum')
).reset_index()

# Score each dimension into quartiles (1=worst, 4=best)
rfm['R_Score'] = pd.qcut(rfm['Recency'], 4, labels=[4,3,2,1])   # lower recency = better
rfm['F_Score'] = pd.qcut(rfm['Frequency'].rank(method='first'), 4, labels=[1,2,3,4])
rfm['M_Score'] = pd.qcut(rfm['Monetary'], 4, labels=[1,2,3,4])

rfm['RFM_Score'] = (rfm['R_Score'].astype(int) + rfm['F_Score'].astype(int) + rfm['M_Score'].astype(int))

# Segment based on total score
conditions = [rfm['RFM_Score']>=10, rfm['RFM_Score']>=7, rfm['RFM_Score']>=4]
choices    = ['Champions', 'Loyal', 'At Risk']
rfm['CustomerSegment'] = np.select(conditions, choices, default='Lost')

print(rfm['CustomerSegment'].value_counts())
print(rfm.nlargest(5, 'Monetary')[['CustID','Recency','Frequency','Monetary','CustomerSegment']])
