Styles & Themes — Instant Visual Overhaul

Built-in style sheets that transform your chart's entire look in one line

Matplotlib ships with dozens of built-in style sheets that change fonts, colors, gridlines, and backgrounds all at once. Instead of manually styling every element, apply a theme globally.

PYTHON
import matplotlib.pyplot as plt

# See all available styles
print(plt.style.available)
# ['default', 'classic', 'ggplot', 'seaborn-v0_8', 'seaborn-v0_8-darkgrid',
#  'fivethirtyeight', 'bmh', 'dark_background', 'tableau-colorblind10', ...]

# Apply a style GLOBALLY — affects every plot after this line
plt.style.use('ggplot')
fig, ax = plt.subplots()
ax.plot([1,2,3], [4,5,6])

# Reset to default
plt.style.use('default')

# Apply a style TEMPORARILY, only within the 'with' block
with plt.style.context('dark_background'):
    fig, ax = plt.subplots()
    ax.plot([1,2,3], [1,4,9])
# Style reverts back automatically after the 'with' block

rcParams — Global Default Settings

PYTHON
# rcParams controls EVERY default setting Matplotlib uses
plt.rcParams['figure.figsize'] = (10, 6)     # default size for ALL figures
plt.rcParams['font.size'] = 12              # default font size
plt.rcParams['axes.grid'] = True              # gridlines on by default
plt.rcParams['lines.linewidth'] = 2          # default line thickness
plt.rcParams['axes.spines.top'] = False       # hide top border by default
plt.rcParams['axes.spines.right'] = False     # hide right border by default

# Reset all rcParams back to Matplotlib defaults
plt.rcdefaults()

# Set multiple at once
plt.rcParams.update({
    'font.family': 'sans-serif',
    'axes.titlesize': 16,
    'axes.labelsize': 12,
    'legend.fontsize': 10
})

Colors & Colormaps

Choosing colors that communicate meaning, not just decoration

Specifying Colors — 5 Ways

PYTHON
# 1. Named colors
ax.plot(x, y, color='steelblue')

# 2. Hex codes
ax.plot(x, y, color='#3B82F6')

# 3. RGB tuple (0-1 scale)
ax.plot(x, y, color=(0.2, 0.4, 0.9))

# 4. RGBA tuple (with transparency)
ax.plot(x, y, color=(0.2, 0.4, 0.9, 0.5))

# 5. Grayscale as a string of a number 0-1
ax.plot(x, y, color='0.5')   # mid-gray

# Get colors from the current style cycle
colors = plt.rcParams['axes.prop_cycle'].by_key()['color']
print(colors)   # ['#1f77b4', '#ff7f0e', '#2ca02c', ...] default matplotlib cycle

PYTHON
# Applying colormaps
sc = ax.scatter(x, y, c=values, cmap='viridis')
fig.colorbar(sc, ax=ax, label='Value')

# Reversed colormap — append '_r'
ax.scatter(x, y, c=values, cmap='viridis_r')

# Get discrete colors from a colormap for categorical data
import matplotlib.pyplot as plt
cmap = plt.get_cmap('tab10')
categories = ['A','B','C','D']
for i, cat in enumerate(categories):
    ax.bar(cat, 10+i, color=cmap(i))   # cmap(i) returns an RGBA tuple
⚠️ Avoid 'jet' and Rainbow Colormaps
Rainbow colormaps like 'jet' are perceptually non-uniform — they create false visual boundaries and are not colorblind-friendly. Prefer viridis, plasma, or cividis, which are perceptually uniform and accessible.

GridSpec — Complex Subplot Layouts

Beyond simple grids — plots that span multiple rows or columns

Regular plt.subplots(nrows, ncols) only creates uniform grids. GridSpec lets you build irregular layouts — like a large main chart with smaller supporting charts around it.

PYTHON
import matplotlib.pyplot as plt
import matplotlib.gridspec as gridspec

fig = plt.figure(figsize=(12, 8))
gs = fig.add_gridspec(3, 3)   # 3 rows, 3 columns of "cells"

# A big chart spanning the top 2 rows, all 3 columns
ax_main = fig.add_subplot(gs[0:2, :])
ax_main.plot([1,2,3], [1,4,9])
ax_main.set_title('Main Chart (spans top, all columns)')

# 3 small charts in the bottom row
ax1 = fig.add_subplot(gs[2, 0])
ax2 = fig.add_subplot(gs[2, 1])
ax3 = fig.add_subplot(gs[2, 2])
ax1.bar(['A','B'], [3,5])
ax2.hist([1,2,2,3,3,3])
ax3.scatter([1,2,3], [3,1,2])

fig.tight_layout()

# Custom row/column size ratios
gs2 = fig.add_gridspec(2, 2,
                        width_ratios=[2, 1],    # left column 2x wider than right
                        height_ratios=[1, 2],   # bottom row 2x taller than top
                        hspace=0.3, wspace=0.3)   # spacing between cells

# mosaic — an even simpler modern way (Matplotlib 3.3+)
fig, axes = plt.subplot_mosaic("""
    AAB
    AAC
    DEF
""", figsize=(10,7))
axes['A'].set_title('Large plot A')
axes['B'].set_title('B')
axes['F'].set_title('F')
💡 subplot_mosaic — The Modern Approach
plt.subplot_mosaic() uses an ASCII-art-like string to define layout — repeated letters mean "this Axes spans these cells". It's often more intuitive than GridSpec's slice notation for complex dashboards.

Typography & Text Styling

Fonts, sizes, weights, and mathematical notation

PYTHON
# Full font control on any text element
ax.set_title('Sales Report',
             fontsize=18,
             fontweight='bold',        # 'normal', 'bold', 'light'
             fontstyle='italic',       # 'normal', 'italic', 'oblique'
             fontfamily='serif',       # 'serif', 'sans-serif', 'monospace'
             color='#1E3A8A',
             loc='left')               # 'left', 'center', 'right'

# LaTeX-style mathematical notation (enclose in $...$)
ax.set_xlabel(r'$x$ (radians)')
ax.set_ylabel(r'$\sin(\theta)$')
ax.set_title(r'$y = \frac{1}{2}at^2 + v_0t$')

# Text with background box
ax.text(0.5, 0.9, 'Important Note',
        transform=ax.transAxes,      # coordinates as 0-1 fraction of axes, not data values
        fontsize=12,
        bbox=dict(boxstyle='round', facecolor='wheat', alpha=0.5))

# Global font settings for the whole figure
plt.rcParams['font.family'] = 'DejaVu Sans'
plt.rcParams['font.size'] = 11

Shared Axes & Advanced Subplot Options

Aligning multiple charts for easy comparison

PYTHON
# Shared x-axis — great for stacked time series (e.g., price + volume)
fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(8,6), sharex=True)
ax1.plot(dates, prices)
ax1.set_ylabel('Price')
ax2.bar(dates, volumes)
ax2.set_ylabel('Volume')
# Only the bottom chart shows x-axis tick labels — cleaner look
fig.subplots_adjust(hspace=0)   # no vertical gap between them

# Shared y-axis — useful for comparing distributions side by side
fig, axes = plt.subplots(1, 3, figsize=(12,4), sharey=True)
axes[0].hist(class_a)
axes[1].hist(class_b)
axes[2].hist(class_c)
# All 3 share the same y-scale for direct visual comparison

# Iterate over axes cleanly with flatten() for a grid
fig, axes = plt.subplots(2, 3, figsize=(14,8))
data_sets = [d1, d2, d3, d4, d5, d6]
for ax, data, title in zip(axes.flatten(), data_sets, titles):
    ax.hist(data)
    ax.set_title(title)
fig.tight_layout()

Inset Axes & Zoomed Views

Adding a mini chart inside a larger chart

PYTHON
# Add a small inset axes INSIDE the main plot — great for zoomed-in detail
fig, ax = plt.subplots(figsize=(8,6))
ax.plot(x, y)
ax.set_title('Full View with Zoomed Inset')

# inset position: [left, bottom, width, height] as fraction of parent axes
ax_inset = ax.inset_axes([0.6, 0.6, 0.35, 0.35])
ax_inset.plot(x, y)
ax_inset.set_xlim(2, 4)     # zoom into a specific x-range
ax_inset.set_ylim(-1, 1)
ax_inset.set_title('Zoomed', fontsize=9)

# Draw a box on the main plot showing what area is zoomed
ax.indicate_inset_zoom(ax_inset, edgecolor='gray')
plt.style.use()

import matplotlib.pyplot as plt
import numpy as np

# Nimbus Analytics brand colors
BRAND = {
    'primary'  : '#6D28D9',
    'secondary': '#0891B2',
    'accent'   : '#F59E0B',
    'dark'     : '#1E1B4B',
    'light'    : '#F5F3FF'
}

# Global rcParams setup
plt.rcParams.update({
    'figure.figsize': (10,6),
    'font.family': 'sans-serif',
    'font.size': 11,
    'axes.spines.top': False,
    'axes.spines.right': False,
    'axes.edgecolor': BRAND['dark'],
    'axes.titlesize': 15,
    'axes.titleweight': 'bold',
    'axes.grid': True,
    'grid.alpha': 0.25

# Sample business data
np.random.seed(5)
quarters = ['Q1','Q2','Q3','Q4']

Design the Layout with subplot_mosaic

fig, axes = plt.subplot_mosaic("""
