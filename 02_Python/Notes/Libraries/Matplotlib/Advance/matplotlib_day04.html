FuncAnimation — The Core Animation Engine

How Matplotlib animates: redraw a frame, repeat

FuncAnimation works by repeatedly calling a function you define, each time updating the plot data for the next "frame". Matplotlib handles the redraw timing and looping.

🎬 The Animation Pattern
1. Create the figure and initial (empty) plot ONCE
2. Define update(frame) — modifies the plot data for that frame
3. FuncAnimation(fig, update, frames=N, interval=ms) — runs it
Only update the DATA inside update() — never recreate the whole figure each frame (very slow).
PYTHON
import matplotlib.pyplot as plt
import matplotlib.animation as animation
import numpy as np

fig, ax = plt.subplots(figsize=(8,5))
x = np.linspace(0, 2*np.pi, 100)

# Create the line ONCE — we'll update its data each frame
line, = ax.plot(x, np.sin(x))
ax.set_ylim(-1.5, 1.5)
ax.set_title('Animated Sine Wave')

def update(frame):
    # frame goes 0, 1, 2, ... automatically
    y = np.sin(x + frame * 0.1)   # shift the wave slightly each frame
    line.set_ydata(y)              # update ONLY the data — fast!
    return line,                  # must return an iterable of updated artists

ani = animation.FuncAnimation(
    fig, update,
    frames=100,        # total number of frames
    interval=50,        # milliseconds between frames (50ms = 20fps)
    blit=True          # only redraw changed parts — much faster
)

plt.show()
💡 blit=True — The Performance Secret
With blit=True, Matplotlib only redraws the parts of the figure that actually CHANGED (the line data), not the entire canvas (axes, labels, gridlines). This is dramatically faster, especially for complex plots. The tradeoff: your update function must return the changed artists, and it works best on simpler animations.

FuncAnimation Parameters

ParameterPurpose
figThe Figure object to animate
funcUpdate function called each frame
framesNumber of frames, or an iterable/generator of frame values
intervalDelay between frames in milliseconds
blitOnly redraw changed elements (faster, but has limitations)
repeatWhether to loop the animation (default True)
init_funcOptional function to draw the initial clean state

Animating Common Chart Types

Line, scatter, bar — the patterns for each

Growing Line Chart (Time Series Reveal)

PYTHON
fig, ax = plt.subplots(figsize=(8,5))
dates = np.arange(100)
prices = 100 + np.cumsum(np.random.randn(100))

line, = ax.plot([], [], color='#C2410C', linewidth=2)
ax.set_xlim(0, 100)
ax.set_ylim(prices.min()-5, prices.max()+5)
ax.set_title('Stock Price — Building Up')

def update(frame):
    line.set_data(dates[:frame], prices[:frame])   # reveal more data each frame
    return line,

ani = animation.FuncAnimation(fig, update, frames=len(dates), interval=30, blit=True)

Animated Scatter Plot

PYTHON
fig, ax = plt.subplots(figsize=(7,7))
n = 50
scat = ax.scatter(np.random.rand(n), np.random.rand(n), c='#F97316', s=50)
ax.set_xlim(0,1); ax.set_ylim(0,1)

def update(frame):
    # Random walk: nudge every point slightly
    new_offsets = scat.get_offsets() + np.random.normal(0, 0.01, (n,2))
    new_offsets = np.clip(new_offsets, 0, 1)      # keep points in bounds
    scat.set_offsets(new_offsets)         # update position data
    return scat,

ani = animation.FuncAnimation(fig, update, frames=200, interval=40, blit=True)

Animated Bar Chart (Race Chart Style)

PYTHON
fig, ax = plt.subplots(figsize=(8,5))
categories = ['A','B','C','D']
# Simulated growth data over 30 "months"
growth = np.cumsum(np.random.rand(30, 4), axis=0)

bars = ax.bar(categories, growth[0], color=['#C2410C','#F97316','#FDBA74','#FED7AA'])
ax.set_ylim(0, growth.max()*1.1)

def update(frame):
    for bar, height in zip(bars, growth[frame]):
        bar.set_height(height)     # update each bar's height
    ax.set_title(f'Month {frame+1}')
    return bars

# NOTE: when the title changes each frame, blit=True can't fully cover it —
# often set blit=False for animations with changing text
ani = animation.FuncAnimation(fig, update, frames=30, interval=150, blit=False)

Interactive Widgets — Sliders & Buttons

Let users control parameters live, without re-running code

The matplotlib.widgets module provides interactive UI elements — sliders, buttons, checkboxes, radio buttons — that work within a plot window (Jupyter with an interactive backend, or a standalone window).

PYTHON
from matplotlib.widgets import Slider, Button, RadioButtons

fig, ax = plt.subplots(figsize=(8,6))
plt.subplots_adjust(bottom=0.35)   # make room at the bottom for widgets

x = np.linspace(0, 10, 200)
init_freq = 1.0
line, = ax.plot(x, np.sin(init_freq * x))
ax.set_ylim(-1.5, 1.5)

# ── SLIDER ── position: [left, bottom, width, height]
ax_freq = plt.axes([0.25, 0.20, 0.5, 0.03])
freq_slider = Slider(ax_freq, 'Frequency', 0.1, 5.0, valinit=init_freq)

def update(val):
    line.set_ydata(np.sin(freq_slider.val * x))
    fig.canvas.draw_idle()      # efficient redraw

freq_slider.on_changed(update)   # call update() whenever the slider moves

# ── BUTTON ──
ax_reset = plt.axes([0.8, 0.05, 0.1, 0.04])
reset_button = Button(ax_reset, 'Reset')

def reset(event):
    freq_slider.reset()

reset_button.on_clicked(reset)

# ── RADIO BUTTONS ──
ax_radio = plt.axes([0.025, 0.5, 0.15, 0.15])
radio = RadioButtons(ax_radio, ('sin', 'cos', 'tan'))

def change_function(label):
    funcs = {'sin': np.sin, 'cos': np.cos, 'tan': np.tan}
    line.set_ydata(funcs[label](freq_slider.val * x))
    fig.canvas.draw_idle()

radio.on_clicked(change_function)

plt.show()
⚠️ Widgets Need an Interactive Backend
Interactive widgets require a GUI backend (TkAgg, Qt5Agg) or %matplotlib widget in Jupyter. They will NOT work with the default inline/static backend used for embedding plots in reports — always test in the actual target environment.

Event Handling — Clicks, Hovers & Keys

Making charts respond directly to mouse and keyboard

PYTHON
fig, ax = plt.subplots()
ax.scatter(np.random.rand(20), np.random.rand(20))

# Respond to mouse clicks
def on_click(event):
    if event.inaxes:    # only if click was INSIDE the axes
        print(f"Clicked at: ({event.xdata:.2f}, {event.ydata:.2f})")
        ax.plot(event.xdata, event.ydata, 'r*', markersize=15)
        fig.canvas.draw()

fig.canvas.mpl_connect('button_press_event', on_click)

# Respond to mouse movement (hover)
def on_hover(event):
    if event.inaxes:
        ax.set_title(f"Mouse at ({event.xdata:.2f}, {event.ydata:.2f})")
        fig.canvas.draw_idle()

fig.canvas.mpl_connect('motion_notify_event', on_hover)

# Respond to key presses
def on_key(event):
    if event.key == 'c':
        ax.cla()               # clear axes when 'c' is pressed
        fig.canvas.draw()

fig.canvas.mpl_connect('key_press_event', on_key)
Event NameTriggers On
'button_press_event'Mouse click down
'button_release_event'Mouse click released
'motion_notify_event'Mouse movement (hover)
'key_press_event'Keyboard key pressed
'pick_event'Clicking directly on a plotted artist (with picker=True set)

Exporting Animations — GIF & MP4

Sharing your animated charts outside of Python

PYTHON
# Save as GIF (requires 'pillow' — usually already installed with matplotlib)
ani.save('animation.gif', writer='pillow', fps=20)

# Save as MP4 (requires ffmpeg installed separately on your system)
ani.save('animation.mp4', writer='ffmpeg', fps=30, dpi=150)

# More control over the writer
from matplotlib.animation import PillowWriter
writer = PillowWriter(fps=15)
ani.save('smooth_animation.gif', writer=writer)

# Display inline in Jupyter as HTML5 video
from IPython.display import HTML
HTML(ani.to_jshtml())      # interactive JS player with play/pause controls
HTML(ani.to_html5_video())  # embedded HTML5 video
💡 Export Format Guide
  • GIF → universal, works everywhere (Slack, docs, web), but larger file size, no audio, limited colors
  • MP4 → smaller file size, smoother playback, requires ffmpeg installed separately
  • to_jshtml() → best for Jupyter notebooks — gives interactive play/pause/scrub controls

Real-World Task — Animated Epidemic Simulation Dashboard

Estimated time: 90–110 minutes  |  Difficulty: ⭐⭐⭐⭐⭐

📋 Project Brief

You are a Data Visualization Engineer at PublicHealth Labs. Build an animated SIR (Susceptible-Infected-Recovered) epidemic model simulation with an interactive parameter dashboard, exported as a shareable GIF for public communication.

1

Phase 1 — SIR Model Simulation

~25 min▼
1
Implement the SIR Model
STARTER CODE
import matplotlib.pyplot as plt
import matplotlib.animation as animation
import numpy as np

# SIR model parameters
N = 1000          # total population
beta = 0.3       # infection rate
gamma = 0.1      # recovery rate
days = 160

# Initial state
S, I, R = [N-1], [1], [0]

# Simulate day by day using simple Euler method
for t in range(days):
    new_infections = beta * S[-1] * I[-1] / N
    new_recoveries = gamma * I[-1]
    S.append(S[-1] - new_infections)
    I.append(I[-1] + new_infections - new_recoveries)
    R.append(R[-1] + new_recoveries)

S, I, R = np.array(S), np.array(I), np.array(R)
print(f"Peak infections: {I.max():.0f} on day {I.argmax()}")

Phase 2 — Build the Animated Chart

~35 min▼
1
Animate the 3 Curves Growing Over Time
CODE
fig, ax = plt.subplots(figsize=(10,6))
ax.set_xlim(0, days)
ax.set_ylim(0, N*1.05)
ax.set_xlabel('Days'); ax.set_ylabel('People')

line_s, = ax.plot([], [], color='#3B82F6', linewidth=2.5, label='Susceptible')
line_i, = ax.plot([], [], color='#EF233C', linewidth=2.5, label='Infected')
line_r, = ax.plot([], [], color='#059669', linewidth=2.5, label='Recovered')
day_text = ax.text(0.02, 0.95, '', transform=ax.transAxes, fontsize=12, fontweight='bold')
ax.legend(loc='upper right')
ax.set_title('SIR Epidemic Model Simulation')

t_axis = np.arange(len(S))

def update(frame):
    line_s.set_data(t_axis[:frame], S[:frame])
    line_i.set_data(t_axis[:frame], I[:frame])
    line_r.set_data(t_axis[:frame], R[:frame])
    day_text.set_text(f'Day {frame}')
    return line_s, line_i, line_r, day_text

ani = animation.FuncAnimation(fig, update, frames=len(t_axis),
                              interval=30, blit=True, repeat=False)
