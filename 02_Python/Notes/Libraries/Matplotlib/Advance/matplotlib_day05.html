Working with Images

Reading, displaying, and combining images with charts

PYTHON
import matplotlib.pyplot as plt
import matplotlib.image as mpimg
import numpy as np

# Read an image file into a NumPy array
img = mpimg.imread('logo.png')   # shape: (height, width, channels)
print(img.shape)   # e.g. (400, 600, 4) — RGBA

# Display an image
fig, ax = plt.subplots()
ax.imshow(img)
ax.axis('off')          # hide axes/ticks for a clean image display

# Place a logo/watermark INSIDE a chart using inset_axes
fig, ax = plt.subplots(figsize=(8,5))
ax.plot([1,2,3], [4,5,6])
logo_ax = ax.inset_axes([0.85, 0.85, 0.12, 0.12])
logo_ax.imshow(img)
logo_ax.axis('off')

# Generate synthetic "image" data (e.g., for heatmaps of pixel-like data)
noise_img = np.random.rand(100,100,3)   # random RGB image array
fig, ax = plt.subplots()
ax.imshow(noise_img)

# Save the current figure AS an image and re-read it (round trip)
fig.savefig('chart.png', dpi=150)
reloaded = mpimg.imread('chart.png')

Advanced Layout Composition

Combining every layout tool for real dashboard construction

Real dashboards combine GridSpec/mosaic, shared axes, varying panel sizes, and consistent styling — all working together. This section shows the complete pattern.

PYTHON
# A realistic multi-panel dashboard skeleton
fig = plt.figure(figsize=(16, 10))
gs = fig.add_gridspec(3, 4,
                        height_ratios=[0.15, 1, 1],   # thin header row on top
                        hspace=0.45, wspace=0.35)

# Row 0: KPI number cards (thin strip across the top)
kpi_axes = [fig.add_subplot(gs[0, i]) for i in range(4)]
kpi_values = ['₹4.2Cr', '12,450', '94.2%', '+18%']
kpi_labels = ['Revenue', 'Orders', 'Uptime', 'Growth']
for ax, val, lab in zip(kpi_axes, kpi_values, kpi_labels):
    ax.axis('off')
    ax.text(0.5, 0.7, val, ha='center', fontsize=22, fontweight='bold', transform=ax.transAxes)
    ax.text(0.5, 0.2, lab, ha='center', fontsize=11, color='gray', transform=ax.transAxes)

# Row 1: main trend chart (spans first 3 cols) + a summary panel (last col)
ax_trend   = fig.add_subplot(gs[1, 0:3])
ax_summary = fig.add_subplot(gs[1, 3])

# Row 2: four smaller charts
ax_bar  = fig.add_subplot(gs[2, 0])
ax_pie  = fig.add_subplot(gs[2, 1])
ax_hist = fig.add_subplot(gs[2, 2])
ax_heat = fig.add_subplot(gs[2, 3])

fig.suptitle('Executive Dashboard', fontsize=20, fontweight='bold', y=0.98)
💡 KPI Cards Trick
A common dashboard pattern is a row of "number cards" — big bold numbers with small labels underneath. Build these with ax.axis('off') (hide the plot frame entirely) and ax.text() calls at relative positions — no actual chart data needed, just formatted text.

Design Principles for Data Visualization

What separates professional dashboards from cluttered ones

The Data-Ink Ratio (Edward Tufte)

Maximise the ink used to show DATA, minimise "chartjunk" — decorative elements that don't convey information.

PYTHON
# ❌ Chartjunk-heavy defaults
ax.bar(x, y)
# default: heavy border, gridlines everywhere, 3D-style shadows if enabled

# ✅ Clean, high data-ink-ratio version
ax.bar(x, y, color='#0F172A', width=0.6)
ax.spines['top'].set_visible(False)
ax.spines['right'].set_visible(False)
ax.spines['left'].set_visible(False)     # even left spine can go if gridlines show scale
ax.grid(axis='y', alpha=0.2)              # only horizontal gridlines, very subtle
ax.tick_params(left=False)               # remove tick marks, keep labels

Design Checklist for Dashboards

PrincipleImplementation
One message per chartDon't cram unrelated data into one Axes
Consistent color meaningIf blue = "good" in chart 1, keep it in chart 2
Sort categorical dataBar charts sorted by value, not alphabetically (unless order matters)
Direct labeling over legendsLabel lines directly at their endpoint when possible — faster to read
Appropriate precisionDon't show 4 decimal places if 1 is meaningful
Highlight, don't decorateUse color/bold sparingly to draw attention to the KEY insight
PYTHON
# Direct labeling technique — label lines at their endpoint instead of a legend
fig, ax = plt.subplots(figsize=(8,5))
for name, series, color in zip(['Product A','Product B'], [series_a, series_b], ['#0F172A','#D97706']):
    ax.plot(months, series, color=color, linewidth=2.5)
    ax.text(len(months)-0.5, series[-1], f' {name}', color=color,
            fontsize=11, fontweight='bold', va='center')
# No legend needed — labels sit right at the line's end, faster to read!

# Highlighting one key series among many gray ones
fig, ax = plt.subplots(figsize=(8,5))
for name, series in all_products.items():
    if name == 'Flagship Product':
        ax.plot(months, series, color='#D97706', linewidth=3, zorder=5, label=name)
    else:
        ax.plot(months, series, color='lightgray', linewidth=1, zorder=1)
ax.legend()

Multi-Page PDF Reports

Combining multiple figures into one shareable document

PYTHON
from matplotlib.backends.backend_pdf import PdfPages

# Combine multiple figures into ONE multi-page PDF file
with PdfPages('quarterly_report.pdf') as pdf:

    # Page 1: title / summary page
    fig1, ax1 = plt.subplots(figsize=(11,8.5))   # US Letter proportions
    ax1.axis('off')
    ax1.text(0.5, 0.6, 'Q4 2024 Report', ha='center', fontsize=28, fontweight='bold')
    ax1.text(0.5, 0.5, 'Prepared by Data Team', ha='center', fontsize=14, color='gray')
    pdf.savefig(fig1)
    plt.close(fig1)

    # Page 2: revenue chart
    fig2, ax2 = plt.subplots(figsize=(11,8.5))
    ax2.plot(months, revenue)
    ax2.set_title('Revenue Trend')
    pdf.savefig(fig2)
    plt.close(fig2)

    # Page 3: dashboard grid
    fig3, axes = plt.subplots(2,2, figsize=(11,8.5))
    # ... populate axes ...
    pdf.savefig(fig3)
    plt.close(fig3)

    # Metadata (visible in PDF properties)
    d = pdf.infodict()
    d['Title']  = 'Quarterly Business Report'
    d['Author'] = 'Data Analytics Team'

print("Multi-page PDF report saved ✅")
✅ PdfPages — The Multi-Chart Report Tool
PdfPages lets you accumulate any number of separate Matplotlib figures into ONE PDF file, each as its own page — perfect for board decks, monthly reports, or scientific papers where each chart deserves its own full page.

np.random.seed(21)

# Monthly revenue for 24 months

base = 80 + months_24 * 2.2
revenue = base + np.random.normal(0, 6, 24) + 15*np.sin(months_24/2)   # seasonal pattern

# Regional performance
regions = ['North','South','East','West','Central']

               np.random.normal(18,4,80), np.random.normal(12,3,80)]

factors = ['AdSpend','FootTraffic','PriceIndex','Revenue']
corr = np.array([[1.00,0.55,-0.20,0.68],
                 [0.55,1.00,-0.15,0.71],
                 [-0.20,-0.15,1.00,-0.35],
                 [0.68,0.71,-0.35,1.00]])

Build a Separate Animated Forecast Reveal

Create a SEPARATE figure (not part of the static dashboard) animating the historical revenue line drawing in, followed by the forecast line extending with a distinct dashed style and shaded uncertainty band.

CODE
fig, ax = plt.subplots(figsize=(10,6))
all_months = np.concatenate([months_24, forecast_months])
ax.set_xlim(0, 30); ax.set_ylim(50, max(revenue.max(),forecast_revenue.max())+15)

line_hist, = ax.plot([], [], color=NAVY, linewidth=2.5)
line_fcst, = ax.plot([], [], color=GOLD, linewidth=2.5, linestyle='--')
ax.axvline(24, color='gray', linestyle=':', alpha=0.5)
ax.set_title('Revenue Trend + 6-Month Forecast')

def update(frame):
    if frame <= 24:
        line_hist.set_data(months_24[:frame], revenue[:frame])
    else:
        line_hist.set_data(months_24, revenue)
        line_fcst.set_data(forecast_months[:frame-24], forecast_revenue[:frame-24])
    return line_hist, line_fcst

ani = animation.FuncAnimation(fig, update, frames=30, interval=80, blit=True, repeat=False)
ani.save('forecast_reveal.gif', writer='pillow', fps=12)
