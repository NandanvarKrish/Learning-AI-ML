def load_training_data_json(filepath=TRAINING_JSON):
    """Load training dataset from JSON file."""
    if not os.path.exists(filepath):
        return None
    try:
        with open(filepath, "r", encoding="utf-8") as f:
            content = json.load(f)
        # Handle both wrapped and raw formats
        if isinstance(content, dict) and "data" in content:
            raw = content["data"]
        else:
            raw = content
        return [(item["message"], item["label"]) for item in raw]
    except (json.JSONDecodeError, KeyError) as e:
        print(f"⚠  Error reading {filepath}: {e}")
        return None


def save_training_data_json(data, filepath=TRAINING_JSON):
    """Save training dataset to JSON with metadata."""
    content = {
        "version": "1.0",
        "description": "NeuroMail Training Dataset",
        "last_updated": datetime.now().isoformat(),
        "count": len(data),
        "data": [{"message": msg, "label": lbl} for msg, lbl in data]
    }
    with open(filepath, "w", encoding="utf-8") as f:
        json.dump(content, f, indent=2, ensure_ascii=False)
    print(f"[+] Saved {len(data)} examples to {filepath}")

def load_training_data_csv(filepath=TRAINING_CSV):
    """Load training dataset from CSV file."""
    if not os.path.exists(filepath):
        return None
    try:
        data = []
        with open(filepath, "r", encoding="utf-8", newline="") as f:
            reader = csv.DictReader(f)
            for row in reader:
                msg = row.get("message", "").strip()
                label = row.get("label", "").strip().lower()
                if msg and label in ("spam", "ham"):
                    data.append((msg, label))
        return data
    except Exception as e:
        print(f"⚠  Error reading {filepath}: {e}")
        return None


def save_training_data_csv(data, filepath=TRAINING_CSV):
    """Save training dataset to CSV."""
    with open(filepath, "w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=["message", "label"])
        writer.writeheader()
        for msg, label in data:
            writer.writerow({"message": msg, "label": label})
    print(f"[+] Exported {len(data)} examples to {filepath}")


def log_prediction(message, label, scores):
    """Append a prediction to the audit log CSV."""
    file_exists = os.path.exists(PREDICTIONS_CSV)
    with open(PREDICTIONS_CSV, "a", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(
            f,
            fieldnames=["timestamp", "message", "prediction", 
                        "spam_score", "ham_score"]
        )
        if not file_exists:
            writer.writeheader()
        writer.writerow({
            "timestamp": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            "message": message,
            "prediction": label,
            "spam_score": round(scores["spam"], 4),
            "ham_score": round(scores["ham"], 4),
        })


def view_prediction_history(limit=10):
    """Display recent predictions from the log."""
    if not os.path.exists(PREDICTIONS_CSV):
        print("\n⚠  No prediction history yet!")
        return
    with open(PREDICTIONS_CSV, "r", encoding="utf-8") as f:
        reader = csv.DictReader(f)
        rows = list(reader)

    print(f"\n📜  Last {min(limit, len(rows))} predictions:")
    print("-" * 80)
    for row in rows[-limit:]:
        emoji = "🚫" if row["prediction"] == "spam" else "✅"
        msg = row["message"][:40]
        print(f"  {emoji} [{row['timestamp']}] {msg:<43} → {row['prediction'].upper()}")
    print("-" * 80)

  • • encoding="utf-8" — handles emojis & special chars
Smart Fallback logic

Smart Data Loader

Tries JSON first → falls back to CSV → creates default if neither exists. Uses Day 3 conditions!

def load_training_data():
    """Load training data from JSON, CSV, or create default."""
    # Try JSON first
    data = load_training_data_json()
    if data:
        print(f"[*] Loaded {len(data)} examples from {TRAINING_JSON}")
        return data

    # Fallback to CSV
    data = load_training_data_csv()
    if data:
        print(f"[*] Loaded {len(data)} examples from {TRAINING_CSV}")
        save_training_data_json(data)  # Auto-save as JSON
        return data

    # Default fallback — create both files
    print("[*] No data files found. Creating default dataset...")
    default_data = [
        ("win a free iphone now click here", "spam"),
        ("hey are we still on for lunch tomorrow", "ham"),
        # ... more examples
    ]
    save_training_data_json(default_data)
    save_training_data_csv(default_data)
    return default_data

ML Core: Train & Predict

The Naive Bayes classifier — uses dictionaries, sets, loops, and functions.

def train_brain(data):
    """Train the Naive Bayes model from labeled data."""
    word_counts = {"spam": {}, "ham": {}}
    class_counts = {"spam": 0, "ham": 0}
    total_words = {"spam": 0, "ham": 0}

    for message, label in data:
        class_counts[label] += 1
        words = tokenize(message)
        for word in words:
            word_counts[label][word] = word_counts[label].get(word, 0) + 1
            total_words[label] += 1

    vocabulary = set(word_counts["spam"].keys()) | set(word_counts["ham"].keys())
    vocab_size = len(vocabulary)

    model = {
        "trained_at": datetime.now().isoformat(),
        "samples": len(data),
        "priors": {c: class_counts[c] / len(data) for c in class_counts},
        "word_probs": {},
        "vocab_size": vocab_size,
        "total_words": total_words,
    }

    for label in ("spam", "ham"):
        model["word_probs"][label] = {}
        for word in vocabulary:
            count = word_counts[label].get(word, 0)
            prob = (count + 1) / (total_words[label] + vocab_size)
            model["word_probs"][label][word] = prob

    return model


def predict(message, model):
    """Predict whether a message is spam or ham."""
    words = tokenize(message)
    scores = {}
    for label in ("spam", "ham"):
        log_prior = math.log(model["priors"][label])
        log_likelihood = sum(
            math.log(model["word_probs"][label].get(
                w, 1 / (model["total_words"][label] + model["vocab_size"])
            ))
            for w in words
        )
        scores[label] = log_prior + log_likelihood

    label = "spam" if scores["spam"] > scores["ham"] else "ham"
    return label, scores


def save_brain(model, filename=BRAIN_FILE):
    with open(filename, "w", encoding="utf-8") as f:
        json.dump(model, f, indent=2, ensure_ascii=False)
    print(f"[+] Brain saved to {filename}")


def load_brain(filename=BRAIN_FILE):
    if not os.path.exists(filename):
        return None
    with open(filename, "r", encoding="utf-8") as f:
        return json.load(f)

Fake Spam Generator

def build_markov_chain(data, label="spam"):
    chain = {}
    for msg, lbl in data:
        if lbl != label:
            continue
        words = tokenize(msg)
        for i in range(len(words) - 1):
            chain.setdefault(words[i], []).append(words[i + 1])
    return chain


def generate_spam(chain, length=10):
    if not chain:
        return "no training data"
    word = random.choice(list(chain.keys()))
    result = [word]
    for _ in range(length - 1):
        if word not in chain:
            break
        word = random.choice(chain[word])
        result.append(word)
    return " ".join(result)

def main():
    print("=" * 60)
    print("   🧠  NEUROMAIL v2.0 - Multi-File AI Brain  🧠")
    print("=" * 60)

    training_data = load_training_data()
    model = load_brain()

    if model:
        trained_at = model.get("trained_at", "unknown")
        samples = model.get("samples", "?")
        print(f"[*] Loaded brain (trained on {samples} samples at {trained_at})")
    else:
        print("[*] Training new brain...")
        model = train_brain(training_data)
        save_brain(model)

    while True:
        print("\n┌──────────────────────────────────────┐")
        print("│  [1] Classify a message              │")
        print("│  [2] Add new training examples       │")
        print("│  [3] Generate fake spam              │")
        print("│  [4] Import training data from CSV   │")
        print("│  [5] Export training data to CSV     │")
        print("│  [6] View prediction history         │")
        print("│  [7] Retrain brain                   │")
        print("│  [8] Quit                            │")
        print("└──────────────────────────────────────┘")

        choice = input("\nChoose an option: ").strip()

        if choice == "1":
            msg = input("Enter message to classify: ")
            label, scores = predict(msg, model)
            emoji = "🚫" if label == "spam" else "✅"
            print(f"\n{emoji}  Prediction: {label.upper()}")
            print(f"    Spam score : {scores['spam']:.4f}")
            print(f"    Ham  score : {scores['ham']:.4f}")
            log_prediction(msg, label, scores)  # Auto-log!
            print("    📝 Logged to predictions.csv")

        elif choice == "2":
            print("\nAdd messages (format: message | spam/ham). Type 'done' to finish.")
            added = 0
            while True:
                line = input("  > ")
                if line.lower() == "done":
                    break
                if "|" not in line:
                    print("    ⚠  Use format: message | spam")
                    continue
                msg, lbl = line.split("|", 1)
                lbl = lbl.strip().lower()
                if lbl not in ("spam", "ham"):
                    print("    ⚠  Label must be 'spam' or 'ham'")
                    continue
                training_data.append((msg.strip(), lbl))
                added += 1
            if added > 0:
                save_training_data_json(training_data)
                save_training_data_csv(training_data)
                print(f"[+] Added {added} examples. Data files updated!")

        elif choice == "3":
            chain = build_markov_chain(training_data, "spam")
            print("\n🤖  Generated spam:")
            for _ in range(3):
                print("   →", generate_spam(chain, length=12))

        elif choice == "4":
            print(f"\n[*] Looking for CSV at: {TRAINING_CSV}")
            new_data = load_training_data_csv()
            if new_data:
                training_data = new_data
                save_training_data_json(training_data)
                print(f"[+] Imported {len(training_data)} examples from CSV!")
                model = train_brain(training_data)
                save_brain(model)
                print("[+] Brain retrained with CSV data.")
            else:
                print(f"⚠  Could not load {TRAINING_CSV}")

        elif choice == "5":
            save_training_data_csv(training_data)
            print(f"\n💡  Open {TRAINING_CSV} in Excel or Google Sheets!")

        elif choice == "6":
            view_prediction_history(10)

        elif choice == "7":
            print("[*] Retraining brain...")
            model = train_brain(training_data)
            save_brain(model)
            print("[+] Brain retrained and saved!")

        elif choice == "8":
            print("Goodbye, Krish! 👋")
            break

        else:
            print("⚠  Invalid choice, try again.")


if __name__ == "__main__":
    main()

        "samples": len(data),
        "priors": {c: class_counts[c] / len(data) for c in class_counts},
        "word_probs": {},
        "vocab_size": vocab_size,
        "total_words": total_words,
    }

    for label in ("spam", "ham"):
        model["word_probs"][label] = {}
        for word in vocabulary:
            count = word_counts[label].get(word, 0)
            prob = (count + 1) / (total_words[label] + vocab_size)
            model["word_probs"][label][word] = prob

    return model


def predict(message, model):
    words = tokenize(message)
    scores = {}
    for label in ("spam", "ham"):
        log_prior = math.log(model["priors"][label])
        log_likelihood = sum(
            math.log(model["word_probs"][label].get(
                w, 1 / (model["total_words"][label] + model["vocab_size"])
            ))
            for w in words
        )
        scores[label] = log_prior + log_likelihood

    label = "spam" if scores["spam"] > scores["ham"] else "ham"
    return label, scores


def save_brain(model, filename=BRAIN_FILE):
    with open(filename, "w", encoding="utf-8") as f:
        json.dump(model, f, indent=2, ensure_ascii=False)
    print(f"[+] Brain saved to {filename}")


def load_brain(filename=BRAIN_FILE):
    if not os.path.exists(filename):
        return None
    with open(filename, "r", encoding="utf-8") as f:
        return json.load(f)


def build_markov_chain(data, label="spam"):
    chain = {}
    for msg, lbl in data:
        if lbl != label:
            continue
        words = tokenize(msg)
        for i in range(len(words) - 1):
            chain.setdefault(words[i], []).append(words[i + 1])
    return chain


def generate_spam(chain, length=10):
    if not chain:
        return "no training data"
    word = random.choice(list(chain.keys()))
    result = [word]
    for _ in range(length - 1):
        if word not in chain:
            break
        word = random.choice(chain[word])
        result.append(word)
    return " ".join(result)
