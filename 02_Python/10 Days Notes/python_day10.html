Application Architecture Pattern

A reference structure you can reuse for future projects

PYTHON
# 1. DATA MODEL — a dataclass representing one unit of your domain
@dataclass
class Task:
    id: int
    title: str
    priority: str = "medium"
    done: bool = False
    created_at: str = field(default_factory=lambda: datetime.now().isoformat())

    def to_dict(self):
        return self.__dict__   # every object instance has __dict__ of its attributes

# 2. LOGGING DECORATOR — wraps actions with a consistent audit trail
def log_action(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        result = func(*args, **kwargs)
        print(f"[LOG] {func.__name__} executed")
        return result
    return wrapper

# 3. BUSINESS LOGIC — a manager class owning the collection and operations
class TaskManager:
    def __init__(self, filepath="tasks.json"):
        self.filepath = filepath
        self.tasks = self._load()

    def _load(self):
        if not os.path.exists(self.filepath):
            return []
        try:
            with open(self.filepath) as f:
                return [Task(**d) for d in json.load(f)]
        except json.JSONDecodeError:
            print("Warning: corrupted task file, starting fresh.")
            return []

    def _save(self):
        with open(self.filepath, "w") as f:
            json.dump([t.to_dict() for t in self.tasks], f, indent=2)

    @log_action
    def add_task(self, title, priority="medium"):
        if priority not in ("low","medium","high"):
            raise InvalidPriorityError(f"Invalid priority: {priority}")
        new_id = max([t.id for t in self.tasks], default=0) + 1
        self.tasks.append(Task(id=new_id, title=title, priority=priority))
        self._save()

    def complete_task(self, task_id):
        for t in self.tasks:
            if t.id == task_id:
                t.done = True
                self._save()
                return
        raise TaskNotFoundError(f"No task with id {task_id}")

    def pending_tasks(self):
        # generator — yields lazily, doesn't build an unnecessary intermediate list
        for t in self.tasks:
            if not t.done:
                yield t
✅ This Structure Scales to Real Projects
The pattern here — dataclass model, private _load/_save persistence methods, a manager class with public methods, custom exceptions for domain-specific errors — is a genuinely reusable structure you'll see in real Python codebases, from small scripts to production web backends.

def print_menu():
    print("""
1. Add task
2. List all tasks
3. List pending tasks only
4. Complete a task
5. Delete a task
6. Show summary
7. Exit
""")

def main():
    manager = TaskManager()
    while True:
        print_menu()
        choice = input("Choose an option: ").strip()
        try:
            if choice == "1":
                title = input("Task title: ")
                priority = input("Priority (low/medium/high): ") or "medium"
                manager.add_task(title, priority)
            elif choice == "7":
                print("Goodbye!")
                break
            # ... handle remaining options ...
        except (InvalidPriorityError, TaskNotFoundError) as e:
            print(f"Error: {e}")

if __name__ == "__main__":
    main()
